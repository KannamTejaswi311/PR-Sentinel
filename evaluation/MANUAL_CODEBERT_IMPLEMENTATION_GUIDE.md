## Executive Overview & Architectural Flow

Under **Frozen Experimental Protocol v1.0** and **Protocol Amendment v1.1**, CodeBERT defect classification operates via a 2-stage neural architecture:
1. **Contextual Token Representation (768-D):** Generated by pretrained `microsoft/codebert-base` Transformer encoder. For each diff hunk, the `[CLS]` token embedding (768-D) and the attention-masked mean-pooled token embedding (768-D) are concatenated to form a **1536-dimensional feature vector**.
2. **Supervised Classification Head (1536 $\to$ 256 $\to$ 2):**
   - Layer 1: Linear projection $1536 \to 256$.
   - Activation: GELU (Gaussian Error Linear Unit).
   - Regularization: Dropout ($p=0.1$).
   - Layer 2: Linear projection $256 \to 2$ (Logits for Class 0 = Clean, Class 1 = Defective).
   - Softmax normalization produces defect probability $P(\text{Defective} \mid \text{Hunk})$.
   - Decision threshold $\tau^*$ is calibrated strictly on validation data via Youden's J statistic ($J = \text{TPR} - \text{FPR}$).

## Stage 5: Train the Supervised Classifier Head

We now train the classification head using the project's existing implementation in `sentinel/codebert_analyzer.py`.

### 5.1 Architecture Explanation
- Input: 1536-dimensional feature vector ($[\text{CLS}] \oplus \text{Mean-Pooled}$).
- Layer 1 (`fc1`): `Linear(in_features=1536, out_features=256, bias=True)`.
- Activation: `GELU()`.
- Dropout: `Dropout(p=0.1)`.
- Layer 2 (`fc2`): `Linear(in_features=256, out_features=2, bias=True)`.
- Loss Function: `CrossEntropyLoss()` with class weighting to balance classes.
- Optimizer: `AdamW(lr=1e-4, weight_decay=0.01)`.

### 5.2 Create the Training Script
Save the following standalone training script to `evaluation/train_codebert_head.py`:

```powershell
@'
"""
PR-Sentinel: CodeBERT Classification Head Trainer
Trains 1536 -> 256 -> 2 MLP on CodeXGLUE representations.
Calibrates threshold on validation split via Youden's J.
Saves checkpoint to sentinel/weights/codebert_classifier.pt.
"""
import json
import logging
import math
from pathlib import Path
import time
import torch
import torch.nn as nn
from torch.utils.data import DataLoader, TensorDataset
from transformers import AutoModel, AutoTokenizer
import numpy as np
from sklearn.metrics import precision_recall_fscore_support, roc_curve

logging.basicConfig(level=logging.INFO, format="%(asctime)s [%(levelname)s] %(message)s")
logger = logging.getLogger("TrainCodeBERTHead")

DEVICE = torch.device("cuda" if torch.cuda.is_available() else "cpu")
SEED = 42
torch.manual_seed(SEED)
np.random.seed(SEED)

class CodeBERTClassificationHead(nn.Module):
    def __init__(self, in_features: int = 1536, hidden_dim: int = 256, num_classes: int = 2, dropout: float = 0.1):
        super().__init__()
        self.fc1 = nn.Linear(in_features, hidden_dim)
        self.activation = nn.GELU()
        self.dropout = nn.Dropout(dropout)
        self.fc2 = nn.Linear(hidden_dim, num_classes)

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        x = self.fc1(x)
        x = self.activation(x)
        x = self.dropout(x)
        return self.fc2(x)

def extract_features(dataset_path: str, tokenizer, encoder, max_samples: int = 3000):
    """Extract 1536-D representation [CLS || Mean-Pool] using genuine CodeBERT."""
    logger.info(f"Extracting features from {dataset_path} (max_samples={max_samples})...")
    features = []
    labels = []
    with open(dataset_path, "r", encoding="utf-8") as f:
        for idx, line in enumerate(f):
            if idx >= max_samples:
                break
            record = json.loads(line)
            code = record["func"]
            target = int(record["target"])

            tokens = tokenizer(code, max_length=256, padding="max_length", truncation=True, return_tensors="pt").to(DEVICE)
            with torch.no_grad():
                out = encoder(**tokens)
                cls_vec = out.last_hidden_state[:, 0, :] # 768
                mask = tokens["attention_mask"].unsqueeze(-1)
                mean_vec = (out.last_hidden_state * mask).sum(dim=1) / mask.sum(dim=1).clamp(min=1e-9)
                vec_1536 = torch.cat([cls_vec, mean_vec], dim=-1).squeeze(0).cpu().numpy()

            features.append(vec_1536)
            labels.append(target)
            if (idx + 1) % 500 == 0:
                logger.info(f"  Processed {idx + 1} samples...")

    return np.array(features, dtype=np.float32), np.array(labels, dtype=np.int64)

def main():
    logger.info(f"Using device: {DEVICE}")
    model_name = "microsoft/codebert-base"
    tokenizer = AutoTokenizer.from_pretrained(model_name)
    encoder = AutoModel.from_pretrained(model_name).to(DEVICE)
    encoder.eval()

    train_path = "evaluation/datasets/codexglue_defect/train.jsonl"
    valid_path = "evaluation/datasets/codexglue_defect/valid.jsonl"
    test_path  = "evaluation/datasets/codexglue_defect/test.jsonl"

    # Extract 1536-D representations
    X_train, y_train = extract_features(train_path, tokenizer, encoder, max_samples=4000)
    X_val, y_val = extract_features(valid_path, tokenizer, encoder, max_samples=1000)
    X_test, y_test = extract_features(test_path, tokenizer, encoder, max_samples=1000)

    train_dataset = TensorDataset(torch.tensor(X_train), torch.tensor(y_train))
    train_loader = DataLoader(train_dataset, batch_size=32, shuffle=True)

    head = CodeBERTClassificationHead().to(DEVICE)
    optimizer = torch.optim.AdamW(head.parameters(), lr=1e-3, weight_decay=0.01)
    criterion = nn.CrossEntropyLoss()

    logger.info("Training classification head for 15 epochs...")
    head.train()
    for epoch in range(1, 16):
        total_loss = 0.0
        for batch_x, batch_y in train_loader:
            batch_x, batch_y = batch_x.to(DEVICE), batch_y.to(DEVICE)
            optimizer.zero_grad()
            logits = head(batch_x)
            loss = criterion(logits, batch_y)
            loss.backward()
            optimizer.step()
            total_loss += loss.item()
        if epoch % 5 == 0 or epoch == 1:
            logger.info(f"Epoch {epoch:02d}/15 - Loss: {total_loss / len(train_loader):.4f}")

    # Calibration on Validation Split via Youden's J
    head.eval()
    with torch.no_grad():
        val_logits = head(torch.tensor(X_val).to(DEVICE))
        val_probs = torch.softmax(val_logits, dim=-1)[:, 1].cpu().numpy()

    fpr, tpr, thresholds = roc_curve(y_val, val_probs)
    j_scores = tpr - fpr
    best_idx = np.argmax(j_scores)
    optimal_threshold = float(thresholds[best_idx])
    best_j = float(j_scores[best_idx])
    logger.info(f"Optimal Threshold (Youden's J): {optimal_threshold:.4f} (J = {best_j:.4f}, TPR={tpr[best_idx]:.4f}, FPR={fpr[best_idx]:.4f})")

    # Evaluate on Untouched Test Split
    with torch.no_grad():
        test_logits = head(torch.tensor(X_test).to(DEVICE))
        test_probs = torch.softmax(test_logits, dim=-1)[:, 1].cpu().numpy()

    test_preds = (test_probs >= optimal_threshold).astype(int)
    p, r, f1, _ = precision_recall_fscore_support(y_test, test_preds, average="binary")
    acc = (test_preds == y_test).mean()
    logger.info(f"Test Set Evaluation: Acc={acc*100:.2f}%, Precision={p*100:.2f}%, Recall={r*100:.2f}%, F1={f1*100:.2f}%")

    # Save Checkpoint
    checkpoint_dir = Path("sentinel/weights")
    checkpoint_dir.mkdir(parents=True, exist_ok=True)
    checkpoint_path = checkpoint_dir / "codebert_classifier.pt"

    payload = {
        "state_dict": head.cpu().state_dict(),
        "threshold": optimal_threshold,
        "is_trained": True,
        "calibration_metadata": {
            "model": model_name,
            "head": "MLP(1536->256->2)",
            "activation": "GELU",
            "dropout": 0.1,
            "calibration_criterion": "Youden_J",
            "threshold": optimal_threshold,
            "trained_on": "CodeXGLUE_defect_detection",
            "random_seed": SEED,
            "training_status": "TRAINED_CHECKPOINT_LOADED",
            "youden_j_score": best_j,
            "val_tpr": float(tpr[best_idx]),
            "val_fpr": float(fpr[best_idx]),
            "test_precision": float(p),
            "test_recall": float(r),
            "test_f1": float(f1),
            "trained_timestamp": time.strftime("%Y-%m-%dT%H:%M:%SZ", time.gmtime()),
        }
    }
    torch.save(payload, str(checkpoint_path))
    logger.info(f"SUCCESS: Saved trained checkpoint to {checkpoint_path}")

if __name__ == "__main__":
    main()
'@ | Out-File -FilePath "evaluation\train_codebert_head.py" -Encoding utf8
```
